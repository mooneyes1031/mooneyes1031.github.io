{{- $page := .page -}} {{- $image := .image -}} {{- $caption := "" -}} {{- with $page.Params.captions -}} {{- $caption = index . $image.Name | default "" -}} {{- end -}} {{- $alt := or $caption (replaceRE "\\.[^.]+$" "" $image.Name) -}} {{- /* 缩略图：生成 360 / 540 / 720 三种 WebP，浏览器按屏幕挑最小够用的 */ -}} {{- $widths := slice 360 540 720 -}} {{- $srcset := slice -}} {{- $thumb := $image -}} {{- range $widths -}} {{- if le . $image.Width -}} {{- $resized := $image.Resize (printf "%dx webp q78" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $resized.RelPermalink $resized.Width) -}} {{- $thumb = $resized -}} {{- end -}} {{- end -}} {{- if not $srcset -}} {{- $resized := $image.Resize (printf "%dx webp q78" $image.Width) -}} {{- $srcset = slice (printf "%s %dw" $resized.RelPermalink $resized.Width) -}} {{- $thumb = $resized -}} {{- end -}} {{- /* 灯箱大图：1600px WebP，只有点开才加载 */ -}} {{- $large := $image -}} {{- $largeWidth := $image.Width -}} {{- if gt $largeWidth 1600 -}} {{- $largeWidth = 1600 -}} {{- end -}} {{- $large = $image.Resize (printf "%dx webp q78" $largeWidth) -}}
